import numpy as np
import matplotlib.pyplot as plt

# 1. Parâmetros da Simulação
m0 = 0.30        # Massa inicial (kg)
ms = 0.15        # Massa seca (kg)
mf = 0.15        # Massa de combustível (kg)
t_burn = 1.5     # Tempo de queima (s)
m_dot = mf / t_burn  # Perda de massa (kg/s)
F = 12.0         # Empuxo (N)
g = 9.81         # Gravidade (m/s²)
k = 0.000344     # Arrasto (kg/m)

# 2. Resolução Numérica da Trajetória (Método de Euler)
dt = 0.01        # Passo de tempo (s)
t_max = 6.0      # Tempo total da simulação (s)
time_steps = int(t_max / dt)

t = np.zeros(time_steps)
h = np.zeros(time_steps)
v = np.zeros(time_steps)

for i in range(1, time_steps):
    t[i] = t[i-1] + dt
    
    # Determinação da massa e empuxo no instante t
    if t[i-1] <= t_burn:
        m_curr = m0 - m_dot * t[i-1]
        F_curr = F
    else:
        m_curr = ms
        F_curr = 0.0
        
    # Força de arrasto
    F_drag = k * (v[i-1] ** 2) * np.sign(v[i-1])
    
    # Aceleração instantânea
    a = (F_curr - m_curr * g - F_drag) / m_curr
    
    # Atualização de velocidade e posição
    v[i] = v[i-1] + a * dt
    h[i] = h[i-1] + v[i-1] * dt
    
    # Interrompe a subida no apogeu (v = 0)
    if t[i-1] > t_burn and v[i] <= 0:
        t = t[:i]
        h = h[:i]
        v = v[:i]
        break

# 3. Gerando e Salvando o Gráfico
plt.figure(figsize=(10, 5), dpi=300)
plt.plot(t, h, label="Altitude (m)", color="#2bbc8a", linewidth=2.5)
plt.axvline(x=t_burn, color="#e74c3c", linestyle="--", label="Fim da Queima (1.5s)")

plt.title("Simulação da Trajetória do Minifoguete", fontsize=14, fontweight="bold")
plt.xlabel("Tempo (s)", fontsize=11)
plt.ylabel("Altitude (m)", fontsize=11)
plt.grid(True, linestyle=":", alpha=0.6)
plt.legend(loc="upper left")

# Marcador do Apogeu
apogeu_idx = np.argmax(h)
plt.scatter(t[apogeu_idx], h[apogeu_idx], color="#e74c3c", zorder=5)
plt.annotate(f"Apogeu: {h[apogeu_idx]:.1f}m", 
             (t[apogeu_idx], h[apogeu_idx]), 
             textcoords="offset points", 
             xytext=(-30, 10), 
             ha="center", 
             fontweight="bold")

plt.tight_layout()
plt.savefig("trajectory_plot.png")
print("Gráfico 'trajectory_plot.png' gerado com sucesso!")
